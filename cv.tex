%%% One CV section: bold label on the left, content on the right.
\newcommand{\cvsection}[2]{%
  \par\noindent
  \begin{minipage}[t]{0.15\linewidth}\raggedright\textbf{#1}\end{minipage}%
  \hspace{0.03\linewidth}%
  \begin{minipage}[t]{0.80\linewidth}\ignorespaces#2\unskip\end{minipage}%
  \par\vspace{5.5mm}}

%%% A line with a date flush right: \dated{what}{when}

%%% it is the last line of a section (a bare \hfill there overhangs the margin).
